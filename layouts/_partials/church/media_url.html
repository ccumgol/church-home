{{/* 미디어 참조값 하나를 실제 공개 URL 문자열로 바꿔 줍니다. 받아들이는 형태 "/media/uploads/poster.png" → Admin(Decap CMS) 업로드 결과 "https://.../x.jpg" → 외부 URL. 그대로 사용 "hero_sky_bg.webp" → 파일명만. assets/media/ 기준 맵 {filename: "x.webp"} → HugoBlox 의 image 오브젝트 표기 assets/ 안에서 찾은 이미지는 너무 크면 자동으로 줄여서 내보냅니다. (교인이 Admin 에서 원본 사진을 그대로 올려도 페이지가 무거워지지 않도록) assets/ 밖(static/)에 있는 파일은 Hugo 가 손댈 수 없으므로 경로만 돌려줍니다. 찾지 못하면 빈 문자열을 돌려주므로, 호출 측에서 `with` 로 분기하면 됩니다. */}} {{- $ref := . -}} {{- $url := "" -}} {{/* 원본보다 크게 쓸 일이 없는 최대 가로 폭 */}} {{- $max_width := 1600 -}} {{/* HugoBlox 스타일의 {filename: ...} 오브젝트도 받아 줍니다. */}} {{- if reflect.IsMap $ref -}} {{- $ref = index $ref "filename" | default "" -}} {{- end -}} {{- with $ref -}} {{- $s := printf "%v" . -}} {{- if or (hasPrefix $s "http://") (hasPrefix $s "https://") -}} {{/* 외부 URL 은 그대로 */}} {{- $url = $s -}} {{- else -}} {{/* assets/ 안에서 찾을 때 쓸 경로. "/media/uploads/x.png" → "media/uploads/x.png" */}} {{- $asset_path := $s -}} {{- if hasPrefix $asset_path "/" -}} {{- $asset_path = strings.TrimPrefix "/" $asset_path -}} {{- else -}} {{- $asset_path = printf "media/%s" $asset_path -}} {{- end -}} {{- with resources.Get $asset_path -}} {{- $img := . -}} {{/* 이미지이고 지나치게 크면 줄입니다. (SVG 등은 .Width 가 없으므로 건너뜀) */}} {{- if and (eq $img.ResourceType "image") (gt $img.Width $max_width) -}} {{- $img = $img.Resize (printf "%dx webp q85" $max_width) -}} {{- end -}} {{- $url = $img.RelPermalink -}} {{- else -}} {{/* assets/ 에 없으면 static/ 에 있다고 보고 경로만 돌려줍니다. */}} {{- if hasPrefix $s "/" -}} {{- $url = $s -}} {{- else -}} {{- $url = printf "/media/%s" $s -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $url -}}

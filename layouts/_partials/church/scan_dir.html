{{/* 폴더 아래의 파일 경로를 재귀적으로 모아 돌려줍니다. 빌드 가드(church/guard.html)가 레이아웃 소스를 검사할 때 씁니다. 받는 값 dir (필수) 프로젝트 기준 폴더 경로. 예: "layouts" exts (필수) 모을 확장자 목록. 예: slice ".html" 돌려주는 값: 파일 경로 문자열 슬라이스 ⚠️ append 에 슬라이스를 그대로 넘기면 중첩되므로, 하위 결과는 range 로 풀어서 담습니다. */}} {{- $dir := .dir -}} {{- $exts := .exts -}} {{- $out := slice -}} {{- with try (os.ReadDir $dir) -}} {{- with .Value -}} {{- range . -}} {{- $p := path.Join $dir .Name -}} {{- if .IsDir -}} {{- range partial "church/scan_dir" (dict "dir" $p "exts" $exts) -}} {{- $out = $out | append . -}} {{- end -}} {{- else if in $exts (path.Ext .Name) -}} {{- $out = $out | append $p -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
